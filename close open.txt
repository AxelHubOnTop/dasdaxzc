local flying = false
local flySpeed = 50
local flyConnection

sections.Movement:Toggle({
    Name = "Fly",
    Default = false,
    Callback = function(state)
        flying = state
        local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
        local root = char:WaitForChild("HumanoidRootPart")
        local cam = workspace.CurrentCamera
        local UIS = game:GetService("UserInputService")

        if flying then
            local bv = Instance.new("BodyVelocity")
            local bg = Instance.new("BodyGyro")
            bv.MaxForce = Vector3.new(9e9, 9e9, 9e9)
            bg.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
            bv.Velocity = Vector3.new(0, 0, 0)
            bv.Parent = root
            bg.Parent = root

            flyConnection = RunService.RenderStepped:Connect(function()
                if char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0 then
                    bv.Velocity =
                        (cam.CFrame.LookVector * flySpeed * (UIS:IsKeyDown(Enum.KeyCode.W) and 1 or 0)) +
                        (cam.CFrame.LookVector * -flySpeed * (UIS:IsKeyDown(Enum.KeyCode.S) and 1 or 0)) +
                        (cam.CFrame.RightVector * -flySpeed * (UIS:IsKeyDown(Enum.KeyCode.A) and 1 or 0)) +
                        (cam.CFrame.RightVector * flySpeed * (UIS:IsKeyDown(Enum.KeyCode.D) and 1 or 0))
                    bg.CFrame = cam.CFrame
                end
            end)
        else
            if flyConnection then flyConnection:Disconnect() flyConnection = nil end
            if root:FindFirstChild("BodyVelocity") then root.BodyVelocity:Destroy() end
            if root:FindFirstChild("BodyGyro") then root.BodyGyro:Destroy() end
        end
    end,
}, "FlyToggle")

            pcall(function()
                Terrain.WaterWaveSize = 0

                Terrain.Decoration = false

sections.Server:Header({ Text = "Quick Actions" })

sections.Server:Button({
    Name = "Server Hop (Random)",
    Callback = function()
        Window:Notify({ Title = "Server", Description = "Finding new server..." })
        local requestFunc = request or http_request or (syn and syn.request) or (fluxus and fluxus.request)
        if requestFunc then
            pcall(function()
                local res = requestFunc({
                    Url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100", game.PlaceId),
                    Method = "GET"
                })
                if res and res.Body then
                    local data = HttpService:JSONDecode(res.Body)
                    if data and data.data then
                        local servers = {}
                        for _, v in pairs(data.data) do
                            if type(v) == "table" and v.playing < v.maxPlayers and v.id ~= game.JobId then
                                table.insert(servers, v.id)
                            end
                        end
                        if #servers > 0 then
                            local randomServer = servers[math.random(1, #servers)]
                            TeleportService:TeleportToPlaceInstance(game.PlaceId, randomServer, LocalPlayer)
                        else
                            TeleportService:Teleport(game.PlaceId, LocalPlayer)
                        end
                    end
                end
            end)
        else
            TeleportService:Teleport(game.PlaceId, LocalPlayer)
        end
    end,
})

ToggleBtn.Position = UDim2.new(0.5, -25, 0.05, 0)
ToggleBtn.Size = UDim2.new(0, 50, 0, 50)
ToggleBtn.AutoButtonColor = true

-- Updated Icon ID
ToggleBtn.Image = "rbxassetid://86949082023913" 

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(0.5, 0)
UICorner.Parent = ToggleBtn

local UIStroke = Instance.new("UIStroke")
UIStroke.Color = Color3.fromRGB(0, 150, 255)
UIStroke.Thickness = 2
UIStroke.Transparency = 0.2
UIStroke.Parent = ToggleBtn

local isDragging = false
local startPos = nil
local dragStart = nil

ToggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = true
        startPos = input.Position
        dragStart = ToggleBtn.Position
    end
end)

ToggleBtn.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        isDragging = false
        if startPos and (input.Position - startPos).Magnitude < 10 then
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
        end
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isDragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
        local delta = input.Position - startPos
        ToggleBtn.Position = UDim2.new(
            dragStart.X.Scale, dragStart.X.Offset + delta.X,
            dragStart.Y.Scale, dragStart.Y.Offset + delta.Y
        )
    end
end)
