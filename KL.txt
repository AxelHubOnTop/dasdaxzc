local HttpService        = game:GetService("HttpService")
local MarketplaceService = game:GetService("MarketplaceService")
local Options = {}

local function GetConfigPath()
    local mapName = "Unknown"
    pcall(function() mapName = MarketplaceService:GetProductInfo(game.PlaceId).Name end)
    mapName = mapName:gsub("[^%w%s]", ""):gsub("%s+", "")
    return "AxelHub/" .. mapName .. ".json"
end

local lastSaveRequest = 0
local function SaveConfig()
    lastSaveRequest = tick()
    local currentRequest = lastSaveRequest
    task.delay(1, function()
        if lastSaveRequest ~= currentRequest then return end
        if not (writefile and makefolder) then return end
        local path = GetConfigPath()
        local folder = path:match("(.+)/")
        if not isfolder(folder) then
            local parts = folder:split("/")
            local current = ""
            for _, part in ipairs(parts) do
                current = current .. part
                if not isfolder(current) then makefolder(current) end
                current = current .. "/"
            end
        end
        writefile(path, HttpService:JSONEncode(Options))
    end)
end

local function LoadConfig()
    if not (readfile and isfile) then return end
    local path = GetConfigPath()
    if isfile(path) then
        local ok, result = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
        if ok then
            for k, v in pairs(result) do Options[k] = v end
        end
    end
end

LoadConfig()

local function refreshBossList()
	BossList = {}
	local bossFolder = workspace:FindFirstChild("Monster") and workspace.Monster:FindFirstChild("Boss")
	if bossFolder then
		local seen = {} local bossLevel = {}
		for _, qd in pairs(QuestManager) do
			if type(qd) == "table" and qd.Mob then
				local m, l = qd.Mob, qd.Level or 0
				if not bossLevel[m] or l > bossLevel[m] then bossLevel[m] = l end
			end
		end
		local tmp = {}
		for _, mob in ipairs(bossFolder:GetChildren()) do
			if mob:IsA("Model") and not seen[mob.Name] then
				seen[mob.Name] = true
				table.insert(tmp, {name=mob.Name, level=bossLevel[mob.Name] or 0})
			end
		end
		table.sort(tmp, function(a, b) return a.level < b.level end)
		for _, v in ipairs(tmp) do table.insert(BossList, v.name) end
	end
	if #BossList == 0 then BossList = {"(No Boss Found)"} end
	if BossDropdown then BossDropdown.Values = BossList end
end

refreshBossList()

local function collectAllTrees()
	allTrees = {}
	local Island = workspace:FindFirstChild("Island")
	if not Island then return end
	local function searchTrees(parent)
		for _, child in pairs(parent:GetChildren()) do
			if (child.Name == "Tree" or child.Name == "Small Tree") and (child:IsA("Folder") or child:IsA("Model")) then
				for _, tree in pairs(child:GetChildren()) do
					if tree.Name == "Dead tree" then continue end
					if tree:IsA("Model") then
						local part = tree.PrimaryPart or tree:FindFirstChildWhichIsA("BasePart")
						if part then table.insert(allTrees, part) end
					end
				end
			end
			if child:IsA("Folder") or child:IsA("Model") then searchTrees(child) end
		end
	end
	searchTrees(Island)
end

-- ==================== Quest Functions ====================
local function getQuestDataForLevel(level)
	local bestQuest, bestLevel = nil, -1
	for questName, questData in pairs(QuestManager) do
		if type(questData) == "table" then
			local questLevel = questData.Level or 0
			local mob        = questData.Mob
			if questData.DailyQuest then continue end
			if SKIP_QUESTS[questName] then continue end
			if mob and questLevel <= level and questLevel > bestLevel then
				bestLevel = questLevel
				bestQuest = {
					questName = questName, mob = mob,
					questArgs = {"take", questName},
					amount    = questData.Ammount or questData.Amount or 1
				}
			end
		end
	end
	return bestQuest
end

function checklevel()
	local level = lp.PlayerStats.lvl.Value
	local data  = getQuestDataForLevel(level)
	if data then
		MON        = data.mob
		QUEST_ARGS = data.questArgs
		FOLDER     = (data.amount == 1) and "Boss" or "Mon"
	end
end

local function isQuestValidForLevel()
	local currentQuest = lp.CurrentQuest.Value
	if currentQuest == "" then return true end
	local data = getQuestDataForLevel(lp.PlayerStats.lvl.Value)
	if not data then return true end
	return currentQuest == data.questName

local function getPrefix()
	if selectedWeapon == "None" or selectedWeapon == "" then return "FS" end
	local tool = lp.Backpack:FindFirstChild(selectedWeapon) or (lp.Character and lp.Character:FindFirstChild(selectedWeapon))
	if tool then
		local customAnim = tool:GetAttribute("CustomAnim") or ""
		if customAnim:find("Combat") then return "FS"
		elseif customAnim == "" then return "DF"
		else return "SW" end
	end
	return "SW"
end

local function getNearestHRP()
	local closest = nil
	local shortestDistance = math.huge
	local function checkFolder(folder)
		for _, obj in pairs(folder:GetChildren()) do
			if obj and obj:FindFirstChild("HumanoidRootPart") then
				local hrp = obj:FindFirstChild("HumanoidRootPart")
				if hrp then
					local distance = (myRoot.Position - hrp.Position).Magnitude
					if distance < shortestDistance then shortestDistance = distance closest = hrp end
				end
			end
		end
	end
	checkFolder(folder1) checkFolder(folder2) checkFolder(folder3)
	return closest
end

local function getEntities(mobName, folder)
	local found = {}
	for _, mob in ipairs(folder:GetChildren()) do
		local root = mob:FindFirstChild("HumanoidRootPart")
		local hum  = mob:FindFirstChild("Humanoid")
		if root then Entity_Position[mob.Name] = root.CFrame end
		if mob.Name == mobName and hum and hum.Health > 0 then table.insert(found, mob) end
	end
	return found
end

local function farmBossByName(bossName)
	local folder = workspace.Monster:FindFirstChild("Boss")
	if not folder then return end
	for _, mob in ipairs(folder:GetChildren()) do
		if mob.Name == bossName then
			local hum  = mob:FindFirstChildOfClass("Humanoid")
			local root = mob:FindFirstChild("HumanoidRootPart")
			if hum and root and hum.Health > 0 then
				if state.method == "Above" then
					tp(CFrame.lookAt(root.Position + Vector3.new(0,state.aboveDist,0), root.Position))
				else

local seaKingRespawnConn = nil

local function startSeaKingRespawnWatch()
	if seaKingRespawnConn then return end
	seaKingRespawnConn = lp.CharacterAdded:Connect(function()
		if not autoSeaking then
			seaKingRespawnConn:Disconnect()
			seaKingRespawnConn = nil
			return
		end
		if isLootingSeaKing then
			isLootingSeaKing = false
			seaKingLootDone  = false
		end
		task.wait(1.5)
		pcall(function()
			local bFolder = workspace.Monster:FindFirstChild("Boss")
			if not bFolder then return end
			local SEA_MOBS = {"Serpent", "Sea King"}
			for _, mob in ipairs(bFolder:GetChildren()) do
				for _, mobName in ipairs(SEA_MOBS) do
					if mob.Name == mobName then
						local root = mob:FindFirstChild("HumanoidRootPart")
						if root then
							tp(CFrame.new(root.Position + Vector3.new(0, 5, 0)))
							WindUI:Notify({
								Title   = "Sea King",
								Content = "Respawned! Returning to "..mobName,
								Icon    = "map-pin",
								Duration = 3
							})
						end
						return
					end
				end
			end
		end)
	end)
end

local function stopSeaKingRespawnWatch()
	if seaKingRespawnConn then

	isLootingSeaKing = false
	seaKingLootDone  = false

			local function trySummonFarm()
				if not autoSummonFarm then return false end
				local bossData = SUMMON_BOSS_LIST[selectedSummonBoss]
				if not bossData then return false end
				local folder = workspace.Monster:FindFirstChild("Boss")
				if not folder then return false end
				local mob = folder:FindFirstChild(bossData.BossName) or folder:FindFirstChild(bossData.AI_Name)
				if mob then
					local hum  = mob:FindFirstChildOfClass("Humanoid")
					local root = mob:FindFirstChild("HumanoidRootPart")
					if hum and root and hum.Health > 0 then
						if state.method == "Above" then
							tp(CFrame.lookAt(root.Position + Vector3.new(0, state.aboveDist, 0), root.Position))
						else
							tp(root.CFrame * CFrame.new(0, 0, state.behindDist))
						end
						attack(root.CFrame)
						return true
					end
				end
				if tick() >= summonCooldown then
					summonCooldown = tick() + 5
					WindUI:Notify({Title="Summon Boss", Content="Summoning "..bossData.AI_Name.."...", Icon="skull", Duration=2})
					pcall(function() EtcRemote:InvokeServer("QuestSpawnBoss", bossData) end)
				end
				return true
			end

			local function tryBossFarm()
				if not autoBossFarm or #selectedBosses == 0 then return false end
				local folder = workspace.Monster:FindFirstChild("Boss")
				if not folder then return false end
				if bossIndex > #selectedBosses then bossIndex = 1 end
				local targetMob  = nil
				local startIndex = bossIndex
				repeat
					local bossName = selectedBosses[bossIndex]
					for _, mob in ipairs(folder:GetChildren()) do
						if mob.Name == bossName then
							local hum  = mob:FindFirstChildOfClass("Humanoid")
							local root = mob:FindFirstChild("HumanoidRootPart")
							if hum and root and hum.Health > 0 then targetMob = mob break end
						end
					end
					if not targetMob then bossIndex = bossIndex % #selectedBosses + 1 end
				until targetMob or bossIndex == startIndex
				if not targetMob then return false end
				local root = targetMob:FindFirstChild("HumanoidRootPart")
				if root then
					if state.method == "Above" then
						tp(CFrame.lookAt(root.Position + Vector3.new(0,state.aboveDist,0), root.Position))
					else
						tp(root.CFrame * CFrame.new(0,0,state.behindDist))
					end
					attack(root.CFrame)
					local hum = targetMob:FindFirstChildOfClass("Humanoid")
					if hum and hum.Health <= 0 then bossIndex = bossIndex % #selectedBosses + 1 end
				end
				return true
			end

			local function tryDungeon()
				if not autoDungeon then return false end
				local mobFolder = workspace:FindFirstChild("MOB")
				if not mobFolder then return false end
				local targetMob   = nil
				local myRoot2     = myChar:FindFirstChild("HumanoidRootPart")
				local closestDist = math.huge
				for _, mob in ipairs(mobFolder:GetDescendants()) do
					if not mob:IsA("Model") then continue end
					local hum  = mob:FindFirstChildOfClass("Humanoid")
					local root = mob:FindFirstChild("HumanoidRootPart")
					if hum and root and hum.Health > 0 and myRoot2 then
						local dist = (root.Position - myRoot2.Position).Magnitude
						if dist < closestDist then closestDist = dist targetMob = mob end
					end
				end
				if not targetMob then return false end
				local root = targetMob:FindFirstChild("HumanoidRootPart")
				if root then
					if state.method == "Above" then
						tp(CFrame.lookAt(root.Position + Vector3.new(0,state.aboveDist,0), root.Position))
					else
						tp(root.CFrame * CFrame.new(0,0,state.behindDist))
					end
					attack(root.CFrame)
				end
				return true
			end

			local function tryDropFarm()
				if not autoDropFarm then return false end
				local activeDrops = getActiveDrops()
				if #activeDrops == 0 then return false end
				if dropBossIndex > #activeDrops then dropBossIndex = 1 end
				local dropName = activeDrops[dropBossIndex]
				local bossName = DROP_BOSS_MAP[dropName]
				if bossName then
					local killed = farmBossByName(bossName)
					if not killed then dropBossIndex = dropBossIndex % #activeDrops + 1 end
				else
					dropBossIndex = dropBossIndex % #activeDrops + 1
				end
				return true
			end

			local function tryAutoFarm()
				if not state.autofarm then return false end
				checklevel()
				local currentQuest = lp.CurrentQuest.Value
				if not MON or not FOLDER then return false end
				if currentQuest == "" then
					QuestRemote:InvokeServer(unpack(QUEST_ARGS)) task.wait(0.5) return true
				elseif not isQuestValidForLevel() then
					abandonQuest() task.wait(0.5)
					QuestRemote:InvokeServer(unpack(QUEST_ARGS)) task.wait(0.5) return true
				end
				local folder   = workspace.Monster:FindFirstChild(FOLDER)
				if not folder then return false end
				local entities = getEntities(MON, folder)
				if #entities > 0 then
					local mob  = entities[1]
					local root = mob:FindFirstChild("HumanoidRootPart")
					local hum  = mob:FindFirstChild("Humanoid")
					if root and hum and hum.Health > 0 then
						if state.method == "Above" then
							tp(CFrame.lookAt(root.Position + Vector3.new(0,state.aboveDist,0), root.Position))
						else
							tp(root.CFrame * CFrame.new(0,0,state.behindDist))
						end
						attack(root.CFrame)
					end
				else
					local found = false
					pcall(function()
						for _, entity in pairs(EntityProxy:GetChildren()) do
							if entity.Name == MON then
								local pivot = entity.WorldPivot
								tp(CFrame.new(Vector3.new(pivot.X, pivot.Y+5, pivot.Z)))
								found = true break
							end
						end
					end)
					if not found and Entity_Position[MON] then
						tp(CFrame.new(Entity_Position[MON].Position + Vector3.new(0,60,0)))
					end
				end
				return true
			end

			local function tryTreeFarm()
				if not autoTreeFarm then return false end
				if #allTrees == 0 then collectAllTrees() treeIndex = 1 return true end
				if treeIndex > #allTrees then treeIndex = 1 end
				local part = allTrees[treeIndex]
				if part and part.Parent then
					tp(CFrame.new(part.Position + Vector3.new(0,3,0)))
					task.wait(3)
					treeIndex = treeIndex + 1
				else
					table.remove(allTrees, treeIndex)
				end
				return true
			end

			if trySeaKing()    then return end
			if tryGhostShip()  then return end
			if trySummonFarm() then return end
			if tryBossFarm()   then return end
			if tryDungeon()    then return end
			if tryDropFarm()   then return end
			if tryAutoFarm()   then return end
			if tryTreeFarm()   then return end
		end)
	end
end)
