local startTime = os.clock()
local function getTimestamp()
    local elapsed = os.clock() - startTime
    return string.format("[%06.2fs]", elapsed)
end

local function truncate(str, len)
    str = tostring(str)
    if #str > len then
        return str:sub(1, len) .. "... [+" .. (#str - len) .. " chars]"
    end
    return str
end


local isPaused = false
local function addLog(source, chunk)
    if isPaused then return end
    totalCaught += 1
    logCount += 1

    CountBadge.Text = totalCaught .. " caught"

    if logCount > CONFIG.MaxLogs then
        local first = ScrollFrame:FindFirstChildOfClass("Frame")
        if first then first:Destroy() end
        logCount -= 1
    end

    local entry = Instance.new("Frame")
    entry.Size = UDim2.new(1, 0, 0, 0)
    entry.AutomaticSize = Enum.AutomaticSize.Y
    entry.BackgroundColor3 = (totalCaught % 2 == 0)
        and Color3.fromRGB(28, 29, 33)
        or  Color3.fromRGB(32, 34, 37)
    entry.BorderSizePixel = 0
    entry.LayoutOrder = totalCaught
    entry.Parent = ScrollFrame

    local entryPad = Instance.new("UIPadding")
    entryPad.PaddingLeft  = UDim.new(0, 8)
    entryPad.PaddingRight = UDim.new(0, 8)
    entryPad.PaddingTop   = UDim.new(0, 6)
    entryPad.PaddingBottom= UDim.new(0, 6)
    entryPad.Parent = entry


    local header = Instance.new("Frame")
    header.Size = UDim2.new(1, 0, 0, 16)
    header.BackgroundTransparency = 1
    header.Parent = entry

    local numLabel = Instance.new("TextLabel")
    numLabel.Size = UDim2.new(0, 30, 1, 0)
    numLabel.BackgroundTransparency = 1
    numLabel.Font = Enum.Font.GothamBold
    numLabel.Text = "#" .. totalCaught
    numLabel.TextColor3 = CONFIG.AccentColor
    numLabel.TextSize = 11
    numLabel.TextXAlignment = Enum.TextXAlignment.Left
    numLabel.Parent = header

    local tsLabel = Instance.new("TextLabel")
    tsLabel.Size = UDim2.new(0, 90, 1, 0)
    tsLabel.Position = UDim2.new(0, 34, 0, 0)
    tsLabel.BackgroundTransparency = 1
    tsLabel.Font = Enum.Font.Code
    tsLabel.Text = getTimestamp()
    tsLabel.TextColor3 = CONFIG.WarnColor

    tsLabel.TextXAlignment = Enum.TextXAlignment.Left
    tsLabel.Parent = header

    local srcLabel = Instance.new("TextLabel")
    srcLabel.Size = UDim2.new(1, -130, 1, 0)
    srcLabel.Position = UDim2.new(0, 130, 0, 0)
    srcLabel.BackgroundTransparency = 1
    srcLabel.Font = Enum.Font.Gotham

    srcLabel.TextSize = 11

    srcLabel.Parent = header


    local content = Instance.new("TextLabel")
    content.Size = UDim2.new(1, 0, 0, 0)
    content.Position = UDim2.new(0, 0, 0, 20)
    content.AutomaticSize = Enum.AutomaticSize.Y
    content.BackgroundTransparency = 1
    content.Font = Enum.Font.Code
    content.Text = truncate(chunk, CONFIG.TruncateLen)
    content.TextColor3 = CONFIG.TextColor
    content.TextSize = 11
    content.TextXAlignment = Enum.TextXAlignment.Left
    content.TextWrapped = true
    content.Parent = entry


    local copyBtn = Instance.new("TextButton")
    copyBtn.Size = UDim2.new(0, 50, 0, 16)
    copyBtn.Position = UDim2.new(1, -50, 0, 0)
    copyBtn.BackgroundColor3 = Color3.fromRGB(50, 52, 58)
    copyBtn.Font = Enum.Font.Gotham
    copyBtn.Text = "copy"
    copyBtn.TextColor3 = CONFIG.MutedColor
    copyBtn.TextSize = 10
    copyBtn.Parent = header
    Instance.new("UICorner", copyBtn).CornerRadius = UDim.new(0, 3)

    copyBtn.MouseButton1Click:Connect(function()
        if setclipboard then
            setclipboard(chunk)
            copyBtn.Text = "✓"
            copyBtn.TextColor3 = CONFIG.SuccessColor
            task.delay(1.5, function()
                if copyBtn.Parent then
                    copyBtn.Text = "copy"
                    copyBtn.TextColor3 = CONFIG.MutedColor
                end
            end)
        end
    end)


    task.defer(function()
        ScrollFrame.CanvasPosition = Vector2.new(0, math.huge)
    end)

    table.insert(logs, {
        index = totalCaught,
        timestamp = getTimestamp(),
        source = source,
        chunk = chunk
    })
end


getgenv().loadstring = function(chunk, chunkName)
    addLog(chunkName or "loadstring", chunk)
    return originalLoadstring(chunk, chunkName)
end

CopyAllBtn.MouseButton1Click:Connect(function()
    if setclipboard then
        local out = {}
        for _, log in ipairs(logs) do
            table.insert(out, string.format("%s #%d [%s]\n%s\n", log.timestamp, log.index, tostring(log.source), log.chunk))
        end
        setclipboard(table.concat(out, "\n---\n"))
        CopyAllBtn.Text = "✓ Copied"
        task.delay(1.5, function()
            CopyAllBtn.Text = "📋 Copy All"
        end)
    end
end)

    if isDragging and input.UserInputType == Enum.UserInputType.MouseMovement then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
end)
