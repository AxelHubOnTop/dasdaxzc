local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/xHeptc/Kavo-UI-Library/main/source.lua"))()
local Window = Library.CreateLib("Axel Hub", "DarkTheme")
local Tab = Window:NewTab("Main")
local Section = Tab:NewSection("Main")

Section:NewTextBox("Monster Name", "", function(txt)
	Name_s = txt
end)

Section:NewToggle("Toggle Bring Bullet", "", function(state)
    _G.Auto_Bulletss = state
end)

spawn(function()
while task.wait(.1) do
pcall(function()
if _G.Auto_Bulletss then
for i,v in pairs(workspace.Mobs:GetChildren()) do
    if string.find(v.Name, Name_s) then
    if v.Monster.Health > 0 then
selected_mons = v
    elseif v.Monster.Health <= 0 then
v:Destroy()
    end
    end
end

for i,v in pairs(workspace:GetChildren()) do
    if v.Name == "Part" and (v:FindFirstChild("BodyForce") and v:FindFirstChild("TouchInterest")) then
        v.CFrame = selected_mons.Torso.CFrame * CFrame.new(0,-1,1)
    end
end
end
end)
end
end)

Section:NewButton("Set Bring Mob, Send Bullet Position", "", function()
    cframe_set = game.Players.LocalPlayer.Character.HumanoidRootPart.CFrame
end)

Section:NewToggle("Toggle Send Bullet", "", function(state)
    _G.Auto_Bulletssend = state
end)

spawn(function()
while task.wait(.1) do
pcall(function()
if _G.Auto_Bulletssend then

for i,v in pairs(workspace:GetChildren()) do
    if v.Name == "Part" and (v:FindFirstChild("BodyForce") and v:FindFirstChild("TouchInterest")) then
        v.CFrame = cframe_set * CFrame.new(0,-1,1)
end
end
end
end)
end
end)

Section:NewToggle("Toggle Bring Mobs", "", function(state)
    _G.Auto_bringmob = state
end)

spawn(function()
while task.wait(.1) do
    pcall(function()
    if _G.Auto_bringmob then
    
    for i,v in pairs(workspace.Mobs:GetChildren()) do
    if string.find(v.Name, Name_s) then
        
            v.Torso.CFrame = cframe_set * CFrame.new(math.random(1,2),2,math.random(1,2))

    end
end
    end
end)
end
end)
