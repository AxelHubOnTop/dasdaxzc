local function tprint(tbl, indent)
    indent = indent or 0
    local out = string.rep(" ", indent) .. "{\r\n"
    indent += 2
    for k, v in pairs(tbl) do
        out ..= string.rep(" ", indent)
        out ..= (typeof(k) == "number") and ("[" .. k .. "] = ") or ('["' .. k .. '"] = ')
        local t = typeof(v)
        if t == "number" then
            out ..= v .. ",\r\n"
        elseif t == "string" then
            out ..= string.format("%q", v) .. ",\r\n"
        elseif t == "buffer" then
            local escaped = buffer.tostring(v):gsub(".", function(c)
                return string.format("\\%03d", string.byte(c))
            end)
            out ..= 'buffer.fromstring("' .. escaped .. '"),\r\n'
        elseif t == "table" then
            out ..= tprint(v, indent + 2) .. ",\r\n"
        else
            out ..= '"' .. tostring(v) .. '"' .. ",\r\n"
        end
    end
    return out .. string.rep(" ", indent - 2) .. "}"
end

local function listSavedFiles()
    ensureFolder()
    local result = {}
    for _, f in ipairs(listfiles(SAVE_PATH)) do
        local name = f:match("([^/\\]+)%.lua$")
        if name then table.insert(result, name) end
    end
    return result
end

local function timeStrToSecs(str)
    if not str then return nil end
    local m, s = str:match("(%d+):(%d+)")
    return (m and s) and (tonumber(m) * 60 + tonumber(s)) or nil
end

local function splitPath(path)
    local parts = {}
    for seg in path:gmatch("[^%.]+") do table.insert(parts, seg) end
    return parts
end

local function resolveRemote(path)
    local r = game
    for _, seg in ipairs(splitPath(path)) do
        local nxt = r:FindFirstChild(seg)
        if not nxt then
            local ok, val = pcall(function() return r[seg] end)
            if ok and val then nxt = val else return nil end
        end
        r = nxt
    end
    return r
end

local function readStrBuf(buf, offset, len)
    local s = ""
    for i = offset, offset + len - 1 do
        s ..= string.char(buffer.readu8(buf, i))
    end
    return s
end

local function getTowerIdFromBuffer(buf)
    local len = buffer.readu8(buf, 2)
    return readStrBuf(buf, 4, len)
end

local function replaceIdInBuffer(buf, newId)
    local size   = buffer.len(buf)
    local newBuf = buffer.create(size)
    for i = 0, size - 1 do buffer.writeu8(newBuf, i, buffer.readu8(buf, i)) end
    buffer.writeu8(newBuf, 2, #newId)
    for i = 1, #newId do buffer.writeu8(newBuf, 3 + i, string.byte(newId, i)) end
    return newBuf
end

local function snapshotTowers()
    local ids, container = {}, workspace:FindFirstChild("placedTowers")
    if container then
        for _, v in ipairs(container:GetChildren()) do ids[v.Name] = true end
    end
    return ids
end

local function waitForNewTower(before, timeout)
    local waited, container = 0, workspace:FindFirstChild("placedTowers")
    if not container then return nil end
    repeat
        task.wait(0.05); waited += 0.05
        for _, v in ipairs(container:GetChildren()) do
            if not before[v.Name] then return v.Name end
        end
    until waited >= timeout
    return nil
end

local RecordTable     = {}
local hookActive      = false

local function saveToFile()
    if currentFileName == "" then return end
    writefile(
        SAVE_PATH .. currentFileName .. ".lua",
        "local Macro = function()\n  return " .. tprint(RecordTable) .. "\nend\nreturn Macro()"
    )
end

local function ToggleRecord(fileName)
    ensureFolder()
    RecordTable, currentFileName, hookActive = {}, fileName, false
    setreadonly(gg, false)
    gg.__namecall = newcclosure(function(self, ...)
        local method  = getnamecallmethod()
        local results = table.pack(old(self, ...))
        if not hookActive and (method == "FireServer" or method == "InvokeServer") then
            hookActive = true
            local s, a, t = self, { ... }, tick()
            task.defer(function()
                pcall(function()
                    local cls = s.ClassName
                    if (method == "FireServer" and cls == "RemoteEvent") or
                       (method == "InvokeServer" and cls == "RemoteFunction") then
                        table.insert(RecordTable, {
                            type        = cls == "RemoteEvent" and "RemoteEvent" or "RemoteFunction",
                            path        = s:GetFullName(),
                            args        = a,
                            time        = t,
                            sessionTime = getSessionTime(),
                            serverTime  = workspace.DistributedGameTime,
                        })
                    end
                end)
                hookActive = false
            end)
        end
        return table.unpack(results, 1, results.n)
    end)
    setreadonly(gg, true)
end

local function StopRecord()
    setreadonly(gg, false)
    gg.__namecall = newcclosure(function(self, ...) return old(self, ...) end)
    setreadonly(gg, true)
    task.defer(function()
        task.wait(0.2)
        if #RecordTable > 0 then saveToFile() end
        currentFileName, hookActive = "", false
    end)
end

local towerIdMap          = {}

local function StopPlay()
    if playThread then pcall(task.cancel, playThread); playThread = nil end
    pcall(function()
        if Fluent.Options and Fluent.Options.PlayToggle then
            Fluent.Options.PlayToggle:SetValue(false)
        end
    end)
end

local function waitUntilSessionTime(targetSecs, timeout)
    local waited = 0
    repeat
        task.wait(0.1); waited += 0.1
        local cur = timeStrToSecs(getSessionTime())
        if cur and cur >= targetSecs then return true end
    until waited >= (timeout or 60)
    return false
end

local function firePlace(remote, args, timeout)
    local before = snapshotTowers()
    remote:FireServer(table.unpack(args or {}))
    return waitForNewTower(before, timeout or 8)
end

local function fireTower(remote, args, oldId, timeout)
    local newId = towerIdMap[oldId]
    if not newId then
        local waited = 0
        repeat task.wait(0.05); waited += 0.05
        until towerIdMap[oldId] or waited >= (timeout or 5)
        newId = towerIdMap[oldId]
    end
    if newId then
        remote:FireServer(replaceIdInBuffer(args[1], newId), args[2])
        return true
    end
    return false
end

local function Plays(fileName, mode)
    towerIdMap = {}
    local path = SAVE_PATH .. fileName .. ".lua"
    if not isfile(path) then
        Fluent:Notify({ Title = "Load Failed", Content = "ไม่พบไฟล์ '" .. fileName .. "' ใน " .. SAVE_PATH, Duration = 4 })
        StopPlay(); return
    end
    local ok, macroFn = pcall(loadfile, path)
    if not ok or not macroFn then
        Fluent:Notify({ Title = "Parse Error", Content = "โหลด '" .. fileName .. "' ล้มเหลว ไฟล์อาจเสียหาย", Duration = 4 })
        StopPlay(); return
    end
    local Macro = macroFn()
    if type(Macro) ~= "table" or #Macro == 0 then
        Fluent:Notify({ Title = "Empty Macro", Content = "'" .. fileName .. "' ไม่มี events บันทึกไว้", Duration = 4 })
        StopPlay(); return
    end

    Fluent:Notify({
        Title   = "Macro Started",
        Content = ("'%s' — %d events | mode: %s"):format(fileName, #Macro, mode),
        Duration = 4,
    })

    playThread = task.spawn(function()
        local firstSessSec = timeStrToSecs(Macro[1].sessionTime) or 0
        playStartServerTime = workspace.DistributedGameTime
        firstServerTime     = Macro[1].serverTime or 0

        for i, v in ipairs(Macro) do
            if mode == "time" then
                local targetSec = timeStrToSecs(v.sessionTime)
                if targetSec then
                    waitUntilSessionTime(firstSessSec + (targetSec - firstSessSec), 120)
                end
            elseif mode == "servertime" then
                local offset = (v.serverTime or firstServerTime) - firstServerTime
                local target = playStartServerTime + offset
                while workspace.DistributedGameTime < target do task.wait(0.05) end
            end

            local remote = resolveRemote(v.path)
            if not remote then
                warn(("[Macro] #%d ไม่พบ remote: %s"):format(i, tostring(v.path)))
                continue
            end

            local success  = false
            local maxRetry = (mode == "index") and 60 or 1

            for attempt = 1, maxRetry do
                pcall(function()
                    if v.path:find("sync_RELIABLE") then
                        local buf    = v.args and v.args[1]
                        local opcode = buf and buffer.readu8(buf, 0)
                        if opcode == 1 then
                            local newId = firePlace(remote, v.args, 8)
                            if newId then
                                for _, nv in ipairs(Macro) do
                                    if nv.path:find("towers_RELIABLE") and nv.args and nv.args[1] then
                                        local oldId = getTowerIdFromBuffer(nv.args[1])
                                        if not towerIdMap[oldId] then
                                            towerIdMap[oldId] = newId
                                            break
                                        end
                                    end
                                end
                                success = true
                            end
                        else
                            remote:FireServer(table.unpack(v.args or {}))
                            success = true
                        end
                    elseif v.path:find("towers_RELIABLE") then
                        local oldId = getTowerIdFromBuffer(v.args[1])
                        success = fireTower(remote, v.args, oldId, 5)
                        if not success then
                            warn(("[Macro] #%d ไม่พบ ID map สำหรับ %s"):format(i, oldId))
                        end
                    else
                        if v.type == "RemoteEvent" then
                            remote:FireServer(table.unpack(v.args or {}))
                        elseif v.type == "RemoteFunction" then
                            remote:InvokeServer(table.unpack(v.args or {}))
                        end
                        success = true
                    end
                end)

                if success then break end
                if mode == "index" then
                    task.wait(1)
                    if attempt % 10 == 0 then
                        Fluent:Notify({
                            Title   = "Retrying Event",
                            Content = ("Event #%d รอ retry %d/60 ครั้ง"):format(i, attempt),
                            Duration = 2,
                        })
                    end
                end
            end

            if not success and mode == "index" then
                Fluent:Notify({
                    Title   = "Event Failed",
                    Content = ("Event #%d ล้มเหลวหลังครบ 60 retry"):format(i),
                    Duration = 4,
                })
            end
        end

        Fluent:Notify({ Title = "Macro Done", Content = ("'%s' เล่นครบทั้ง %d events แล้ว"):format(fileName, #Macro), Duration = 5 })
        StopPlay()
    end)
end

            local gen   = RoomsUI.container.general
            local final = RoomsUI.container.finalScreen
            if not gen.Visible and not final.Visible then
                UseRemoteJoin("Create Match"); continue
            end
            local titles = gen.content.leftSide.titlesHolder
            local diff   = titles.difficultyText.Text
            local map    = titles.mapName.Text
            local act    = titles.actNumber.Text
            if Options.Operations.Value == "Challenge" then
                if not diff:find("Legends") then UseRemoteJoin("Challenge"); continue end
                if not map:find(Options.Stage.Value) then UseRemoteJoin(Options.Stage.Value); continue end
                if not act:find(Options.ChallengeAct.Value) then selectAct(act, Options.ChallengeAct.Value); continue end
                if final.Visible then
                    RS:WaitForChild("rooms"):WaitForChild("rooms_RELIABLE"):FireServer(buffer.fromstring("\005\n"), {})
                else
                    UseRemoteJoin("CreateFinal")
                end
            elseif Options.Operations.Value == "Story" then
                if not diff:find("Normal") and not diff:find("Nightmare") then UseRemoteJoin("Story"); continue end
                if not map:find(Options.Stage.Value) then UseRemoteJoin(Options.Stage.Value); continue end
                if not act:find(Options.NormalAct.Value) then selectAct(act, Options.NormalAct.Value); continue end
                if not diff:find(Options.Difficulty.Value) then UseRemoteJoin(Options.Difficulty.Value); continue end
                if final.Visible then
                    RS:WaitForChild("rooms"):WaitForChild("rooms_RELIABLE"):FireServer(buffer.fromstring("\005\n"), {})
                else
                    UseRemoteJoin("CreateFinal")
                end
            end
        end
    end)
end
